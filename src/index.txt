// Excel管理类（修复this + 本地缓存持久化）
class ExcelDataManager {
    constructor() {
        this.data = [];
        this.fileName = '';
        this.loadTime = '';
        this.isLoaded = false;
        this.columnMap = {};
    }
    loadExcel(file) {
        var self = this;
        return new Promise(function(resolve, reject) {
            var reader = new FileReader();
            reader.onload = function(ev) {
                try {
                    var data = new Uint8Array(ev.target.result);
                    var workbook = XLSX.read(data, { type: 'array' });
                    var firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                    var jsonData = XLSX.utils.sheet_to_json(firstSheet);
                    if (!jsonData || jsonData.length === 0) {
                        reject(new Error('Excel文件为空或格式不正确'));
                        return;
                    }
                    self.data = jsonData;
                    self.fileName = file.name;
                    self.loadTime = new Date().toLocaleString();
                    self.isLoaded = true;
                    self.columnMap = self.autoDetectColumns();
                    resolve(jsonData);
                } catch (error) {
                    reject(error);
                }
            };
            reader.onerror = function() {
                reject(new Error('读取文件失败'));
            };
            reader.readAsArrayBuffer(file);
        });
    }
    autoDetectColumns() {
        if (this.data.length === 0) return {};
        var firstRow = this.data[0];
        var keys = Object.keys(firstRow);
        var findKey = function(patterns) {
            for (var i = 0; i < patterns.length; i++) {
                var p = patterns[i];
                for (var j = 0; j < keys.length; j++) {
                    var k = keys[j];
                    if (k.indexOf(p) !== -1 || k.toLowerCase().indexOf(p.toLowerCase()) !== -1) {
                        return k;
                    }
                }
            }
            return '';
        };
        return {
            name: findKey(['型号', '名称', 'model', 'name', '相机']),
            senW: findKey(['长边', '宽', 'width', 'sensor_w', 'senW']),
            senH: findKey(['短边', '高', 'height', 'sensor_h', 'senH']),
            focal: findKey(['焦距', 'focal', 'f']),
            distCam: findKey(['相机工作距离', 'cam_dist', '工作距离', '距离']),
            distLight: findKey(['光源工作距离', 'light_dist', '光源距离'])
        };
    }
    getVersionInfo() {
        if (!this.isLoaded) return '未加载';
        return this.fileName + ' | ' + this.loadTime + ' | ' + this.data.length + '条';
    }
    populateSelect(selectElement) {
        selectElement.innerHTML = '<option value="">-- 请选择型号 --</option>';
        if (!this.isLoaded) return;
        var nameKey = this.columnMap.name || Object.keys(this.data[0]);
        for (var i = 0; i < this.data.length; i++) {
            var row = this.data[i];
            var opt = document.createElement('option');
            opt.value = JSON.stringify(row);
            var displayName = row[nameKey] || ('型号' + (i + 1));
            opt.textContent = displayName;
            selectElement.appendChild(opt);
        }
    }
    fillInputs(row) {
        var map = this.columnMap;
        var inputMap = {
            'senW': map.senW,
            'senH': map.senH,
            'focal': map.focal,
            'distCam': map.distCam || '相机工作距离',
            'distLight': map.distLight || '光源工作距离'
        };
        for (var id in inputMap) {
            var colKey = inputMap[id];
            var element = document.getElementById(id);
            if (element && colKey && row[colKey] !== undefined && row[colKey] !== null) {
                element.value = row[colKey];
            }
        }
    }
    // 本地存储保存
    saveToLocal() {
        const saveObj = {
            data: this.data,
            fileName: this.fileName,
            loadTime: this.loadTime,
            columnMap: this.columnMap
        };
        localStorage.setItem("excelCameraData", JSON.stringify(saveObj));
    }
    // 读取本地缓存
    loadFromLocal() {
        const str = localStorage.getItem("excelCameraData");
        if (!str) return false;
        try {
            const obj = JSON.parse(str);
            this.data = obj.data;
            this.fileName = obj.fileName;
            this.loadTime = obj.loadTime;
            this.columnMap = obj.columnMap;
            this.isLoaded = true;
            return true;
        } catch (e) {
            localStorage.removeItem("excelCameraData");
            return false;
        }
    }
    // 清空本地缓存（保留方法但不再自动调用）
    clearLocal() {
        localStorage.removeItem("excelCameraData");
    }
}

function drawArc(ctx, cx, cy, r, start, end, seg) {
    seg = seg || 30;
    ctx.beginPath();
    var first = true;
    var ang = start;
    var step = (end - start) / seg;
    for (var i = 0; i <= seg; i++) {
        var x = cx + Math.cos(ang) * r;
        var y = cy + Math.sin(ang) * r;
        if (first) { ctx.moveTo(x, y); first = false; }
        else { ctx.lineTo(x, y); }
        ang += step;
    }
    ctx.stroke();
}
function drawCone(ctx, canvas, FOV, workLen, lightLen) {
    var baseApexY = 120, baseWorkY = 260;
    var pixDis = baseWorkY - baseApexY;
    var halfRad = deg2rad(FOV / 2);
    var halfWork = pixDis * Math.tan(halfRad);
    var scale = lightLen / workLen;
    var halfLight = halfWork * scale;
    var lightY = baseWorkY - (pixDis * (scale - 1));
    var downOffset = pixDis * (scale - 1);
    var downPlaneY = baseWorkY + downOffset;
    var halfDown = halfWork * scale;

    var pts = [
        {x: 0, y: baseApexY}, {x: -halfWork, y: baseWorkY}, {x: halfWork, y: baseWorkY},
        {x: -halfLight, y: lightY}, {x: halfLight, y: lightY},
        {x: -halfDown, y: downPlaneY}, {x: halfDown, y: downPlaneY}
    ];
    var minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
    for (var i = 0; i < pts.length; i++) {
        var p = pts[i];
        if (p.x < minX) minX = p.x;
        if (p.x > maxX) maxX = p.x;
        if (p.y < minY) minY = p.y;
        if (p.y > maxY) maxY = p.y;
    }
    var safe = 90, pad = 35;
    minX -= safe; maxX += safe; minY -= safe; maxY += safe;
    var cw = (maxX - minX) + pad * 2;
    var ch = (maxY - minY) + pad * 2;
    var ratio = 1;
    if (cw > MAX_W || ch > MAX_H) {
        ratio = Math.min(MAX_W / cw, MAX_H / ch);
        cw = Math.round(cw * ratio);
        ch = Math.round(ch * ratio);
    }
    var shiftX = pad - minX, shiftY = pad - minY;
    var dp = window.devicePixelRatio || 1;
    canvas.width = cw * dp;
    canvas.height = ch * dp;
    canvas.style.width = cw + 'px';
    canvas.style.height = ch + 'px';
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.save();
    ctx.scale(dp * ratio, dp * ratio);
    var apexX = shiftX, apexY = baseApexY + shiftY;
    var workY = baseWorkY + shiftY;
    var upY = lightY + shiftY;
    var downY = downPlaneY + shiftY;

    // 渲染镜头图片
    const imgW = 30;
    const imgH = 95;
    const imgX = apexX - imgW / 2;
    const imgY = apexY - 80;
    if (lensImage.complete) {
        ctx.drawImage(lensImage, imgX, imgY, imgW, imgH);
    } else {
        ctx.fillStyle = "#3498eb";
        ctx.fillRect(apexX - 32, apexY - 55, 64, 50);
        ctx.fillStyle = "#000";
        ctx.font = "bold 22px Arial";
        ctx.textAlign = "center";
        ctx.fillText("CST", apexX, apexY - 23);
        ctx.fillStyle = "#666";
        ctx.fillRect(apexX - 22, apexY - 5, 44, 35);
    }

    var wL = apexX - halfWork, wR = apexX + halfWork;
    var lL = apexX - halfLight, lR = apexX + halfLight;
    var dL = apexX - halfDown, dR = apexX + halfDown;

    ctx.strokeStyle = "#000";
    ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(apexX, apexY); ctx.lineTo(wL, workY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(apexX, apexY); ctx.lineTo(wR, workY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(wL, workY); ctx.lineTo(wR, workY); ctx.stroke();

    ctx.setLineDash([8, 5]);
    ctx.lineWidth = 1.5;
    ctx.beginPath(); ctx.moveTo(wL, workY); ctx.lineTo(lL, upY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(wR, workY); ctx.lineTo(lR, upY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(lL, upY); ctx.lineTo(lR, upY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(wL, workY); ctx.lineTo(dL, downY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(wR, workY); ctx.lineTo(dR, downY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(dL, downY); ctx.lineTo(dR, downY); ctx.stroke();
    ctx.setLineDash([]);

    // 文字右移35防遮挡
    ctx.fillStyle = "#c82423";
    ctx.font = "bold 22px Arial";
    ctx.textAlign = "left";
    ctx.fillText(lightLen.toFixed(0) + " mm", lR + 35, upY + 8);
    ctx.fillStyle = "#666666";
    ctx.fillText(lightLen.toFixed(0) + " mm", dR + 35, downY + 8);

    var arcR = 52;
    var aL = Math.PI + halfRad - Math.PI / 2;
    var aR = Math.PI - halfRad - Math.PI / 2;
    if (aL < aR) aL += Math.PI * 2;
    ctx.strokeStyle = "#0055ff";
    ctx.lineWidth = 2;
    drawArc(ctx, apexX, apexY, arcR, aL, aR);
    // 角度文字左移25
    ctx.fillStyle = "#0055ff";
    ctx.font = "bold 28px Arial";
    ctx.fillText(FOV.toFixed(1) + "°", apexX + arcR - 25, apexY + 18);

    ctx.setLineDash([6, 4]);
    ctx.strokeStyle = "#000";
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    ctx.moveTo(wL, workY); ctx.lineTo(wL, workY + 30);
    ctx.moveTo(wR, workY); ctx.lineTo(wR, workY + 30);
    ctx.moveTo(wL, workY + 15); ctx.lineTo(wR, workY + 15);
    ctx.stroke();
    ctx.setLineDash([]);
    var arr = 7;
    ctx.beginPath();
    ctx.moveTo(wL, workY + 15); ctx.lineTo(wL + arr, workY + 15 - arr);
    ctx.moveTo(wL, workY + 15); ctx.lineTo(wL + arr, workY + 15 + arr);
    ctx.moveTo(wR, workY + 15); ctx.lineTo(wR - arr, workY + 15 - arr);
    ctx.moveTo(wR, workY + 15); ctx.lineTo(wR - arr, workY + 15 + arr);
    ctx.stroke();
    ctx.fillStyle = "#772299";
    ctx.font = "bold 28px Arial";
    ctx.textAlign = "center";
    ctx.fillText(workLen.toFixed(0) + " mm", apexX, workY + 60);
    ctx.restore();
}
function drawSegmentArc(ctx, cx, cy, radius, startRad, endRad, segmentCount) {
    segmentCount = segmentCount || 30;
    ctx.beginPath();
    var first = true;
    var angle = startRad;
    var step = (endRad - startRad) / segmentCount;
    for (var i = 0; i <= segmentCount; i++) {
        var x = cx + Math.cos(angle) * radius;
        var y = cy + Math.sin(angle) * radius;
        if (first) { ctx.moveTo(x, y); first = false; }
        else { ctx.lineTo(x, y); }
        angle += step;
    }
    ctx.stroke();
}
function drawLineCone(ctx, canvas, totalFOV, workLen, lightLen, camDist, lightExtDist) {
    var baseApexY = 80, baseWorkY = 260;
    var pixCamDistance = baseWorkY - baseApexY;
    var halfRad = deg2rad(totalFOV / 2);
    var halfWorkPix = pixCamDistance * Math.tan(halfRad);
    var scale = lightLen / workLen;
    var upOffset = pixCamDistance * (scale - 1);
    var lightRelY = baseWorkY - upOffset;
    var downOffset = pixCamDistance * (scale - 1);
    var downPlaneY = baseWorkY + downOffset;
    var halfDownPix = halfWorkPix * scale;

    var wLx = -halfWorkPix, wRx = halfWorkPix;
    var lLx = -halfWorkPix * scale, lRx = halfWorkPix * scale;
    var dLx = -halfDownPix, dRx = halfDownPix;

    var points = [
        {x: 0, y: baseApexY - 55}, {x: 0, y: baseApexY + 35},
        {x: wLx, y: baseWorkY}, {x: wRx, y: baseWorkY},
        {x: lLx, y: lightRelY}, {x: lRx, y: lightRelY},
        {x: dLx, y: downPlaneY}, {x: dRx, y: downPlaneY},
        {x: 0, y: baseWorkY + 65}
    ];
    var minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
    for (var i = 0; i < points.length; i++) {
        var p = points[i];
        if (p.x < minX) minX = p.x;
        if (p.x > maxX) maxX = p.x;
        if (p.y < minY) minY = p.y;
        if (p.y > maxY) maxY = p.y;
    }
    var textSafeW = 90, textSafeH = 40;
    minX -= textSafeW; maxX += textSafeW; minY -= textSafeH; maxY += textSafeH;
    var pad = 35;
    var canvasW = (maxX - minX) + pad * 2;
    var canvasH = (maxY - minY) + pad * 2;
    var scaleRatio = 1;
    if (canvasW > MAX_W || canvasH > MAX_H) {
        var ratioW = MAX_W / canvasW;
        var ratioH = MAX_H / canvasH;
        scaleRatio = Math.min(ratioW, ratioH);
        canvasW = Math.round(canvasW * scaleRatio);
        canvasH = Math.round(canvasH * scaleRatio);
    }
    var shiftX = pad - minX, shiftY = pad - minY;
    canvas.width = canvasW;
    canvas.height = canvasH;
    ctx.clearRect(0, 0, canvasW, canvasH);
    ctx.save();
    ctx.scale(scaleRatio, scaleRatio);
    var apexX = shiftX, apexY = baseApexY + shiftY;
    var workLineY = baseWorkY + shiftY;
    var upY = lightRelY + shiftY;
    var downY = downPlaneY + shiftY;

    // 镜头图片渲染
    const imgW = 30;
    const imgH = 95;
    const imgX = apexX - imgW / 2;
    const imgY = apexY - 80;
    if (lensImage.complete) {
        ctx.drawImage(lensImage, imgX, imgY, imgW, imgH);
    } else {
        ctx.fillStyle = '#3498eb';
        ctx.fillRect(apexX - 32, apexY - 55, 64, 50);
        ctx.fillStyle = '#ffffff';
        ctx.font = 'bold 22px Arial';
        ctx.textAlign = 'center';
        ctx.fillText('CST', apexX, apexY - 23);
    }

    var realWLx = apexX + wLx, realWRx = apexX + wRx;
    var realLLx = apexX + lLx, realLRx = apexX + lRx;
    var realDLx = apexX + dLx, realDRx = apexX + dRx;

    ctx.strokeStyle = '#000';
    ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(apexX, apexY); ctx.lineTo(realWLx, workLineY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(apexX, apexY); ctx.lineTo(realWRx, workLineY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(realWLx, workLineY); ctx.lineTo(realWRx, workLineY); ctx.stroke();

    ctx.setLineDash([8, 5]);
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 1.5;
    ctx.beginPath(); ctx.moveTo(realWLx, workLineY); ctx.lineTo(realLLx, upY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(realWRx, workLineY); ctx.lineTo(realLRx, upY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(realLLx, upY); ctx.lineTo(realLRx, upY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(realWLx, workLineY); ctx.lineTo(realDLx, downY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(realWRx, workLineY); ctx.lineTo(realDRx, downY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(realDLx, downY); ctx.lineTo(realDRx, downY); ctx.stroke();
    ctx.setLineDash([]);

    // 文字右移防遮挡
    ctx.fillStyle = '#c82423';
    ctx.font = 'bold 22px Arial';
    ctx.textAlign = 'left';
    ctx.fillText(lightLen.toFixed(0) + ' mm', realLRx + 35, upY + 8);
    ctx.fillStyle = '#666666';
    ctx.fillText(lightLen.toFixed(0) + ' mm', realDRx + 35, downY + 8);

    var arcRadius = 52;
    var rotate90 = Math.PI / 2;
    var angL = Math.PI + halfRad - rotate90;
    var angR = Math.PI - halfRad - rotate90;
    var startA = angL, endA = angR;
    if (startA < endA) startA += Math.PI * 2;
    ctx.strokeStyle = '#0055ff';
    ctx.lineWidth = 2;
    drawSegmentArc(ctx, apexX, apexY, arcRadius, startA, endA);
    ctx.fillStyle = '#0055ff';
    ctx.font = 'bold 28px Arial';
    ctx.textAlign = 'left';
    ctx.fillText(totalFOV.toFixed(1) + '°', apexX + arcRadius - 25, apexY + 18);

    ctx.setLineDash([6, 4]);
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    ctx.moveTo(realWLx, workLineY); ctx.lineTo(realWLx, workLineY + 30);
    ctx.moveTo(realWRx, workLineY); ctx.lineTo(realWRx, workLineY + 30);
    ctx.moveTo(realWLx, workLineY + 15); ctx.lineTo(realWRx, workLineY + 15);
    ctx.stroke();
    ctx.setLineDash([]);
    var arrSz = 7;
    ctx.beginPath();
    ctx.moveTo(realWLx, workLineY + 15); ctx.lineTo(realWLx + arrSz, workLineY + 15 - arrSz);
    ctx.moveTo(realWLx, workLineY + 15); ctx.lineTo(realWLx + arrSz, workLineY + 15 + arrSz);
    ctx.moveTo(realWRx, workLineY + 15); ctx.lineTo(realWRx - arrSz, workLineY + 15 - arrSz);
    ctx.moveTo(realWRx, workLineY + 15); ctx.lineTo(realWRx - arrSz, workLineY + 15 + arrSz);
    ctx.stroke();
    ctx.fillStyle = '#772299';
    ctx.font = 'bold 28px Arial';
    ctx.textAlign = 'center';
    ctx.fillText(workLen.toFixed(0) + ' mm', apexX, workLineY + 60);
    ctx.restore();
}
function calculate() {
    var activeBtn = document.querySelector('.tool-toggle-btn.active');
    var tool = activeBtn.dataset.tool;
    document.getElementById('resultCard').style.display = 'block';
    document.getElementById('canvasCard').style.display = 'block';
    if (tool === 'area') {
        var sw = parseFloat(document.getElementById('senW').value);
        var sh = parseFloat(document.getElementById('senH').value);
        var f = parseFloat(document.getElementById('focal').value);
        var dCam = parseFloat(document.getElementById('distCam').value);
        var dLight = parseFloat(document.getElementById('distLight').value);
        if (isNaN(sw) || isNaN(sh) || isNaN(f) || isNaN(dCam) || isNaN(dLight) || sw <= 0 || sh <= 0 || f <= 0 || dCam <= 0 || dLight <= 0) {
            alert('所有输入必须为大于0的数字');
            return;
        }
        var fovH = rad2deg(2 * Math.atan(sw / (2 * f)));
        var fovV = rad2deg(2 * Math.atan(sh / (2 * f)));
        var fovD = rad2deg(2 * Math.atan(Math.hypot(sw, sh) / (2 * f)));
        var wReal = sw * dCam / f;
        var hReal = sh * dCam / f;
        var wLight = sw * (dCam + dLight) / f;
        var hLight = sh * (dCam + dLight) / f;
        document.getElementById('fovH').innerText = fovH.toFixed(2) + ' °';
        document.getElementById('fovV').innerText = fovV.toFixed(2) + ' °';
        document.getElementById('fovD').innerText = fovD.toFixed(2) + ' °';
        document.getElementById('sizeText').innerText = wReal.toFixed(2) + ' mm × ' + hReal.toFixed(2) + ' mm';
        document.getElementById('lightLenW').innerText = wLight.toFixed(2) + ' mm';
        document.getElementById('lightLenH').innerText = hLight.toFixed(2) + ' mm';
        drawCone(ctxH, canH, fovH, wReal, wLight);
        drawCone(ctxV, canV, fovV, hReal, hLight);
    } else {
        var realW = parseFloat(document.getElementById('realW').value);
        var distCam = parseFloat(document.getElementById('distCamLine').value);
        var distLight = parseFloat(document.getElementById('distLightLine').value);
        if (isNaN(realW) || isNaN(distCam) || isNaN(distLight) || realW <= 0 || distCam <= 0 || distLight <= 0) {
            alert('所有输入必须为大于0的有效数字');
            return;
        }
        var halfRad = Math.atan((realW / 2) / distCam);
        var fovH = rad2deg(2 * halfRad);
        var lightW = realW * (distCam + distLight) / distCam;
        document.getElementById('fovHLine').textContent = fovH.toFixed(2) + ' °';
        document.getElementById('sizeText').textContent = realW.toFixed(2) + ' mm';
        document.getElementById('lightSizeLine').textContent = lightW.toFixed(2) + ' mm';
        drawLineCone(ctxLine, canLine, fovH, realW, lightW, distCam, distLight);
    }
}
